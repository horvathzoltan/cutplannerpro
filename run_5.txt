1. LÉPÉS — A .ui fájlban hozz létre 3 üres widgetet a groupBox_productType‑ban

A Qt Designerben:

A groupBox_productType belsejébe rakd be:

1.  typeContainerWidget
2.  stackedWidget_stackSubtype (már létezik → csak áthelyezed ide)
3.  calcModeContainerWidget

Majd:

-   állítsd a groupBox_productType layoutját QVBoxLayout‑ra
-   a három widget egymás alatt legyen

Ez lesz a három „csík”.

------------------------------------------------------------------------

2. LÉPÉS — A konstruktorban NE hozz létre layoutot közvetlenül a groupBox_productType‑ra

Törlendő:

    auto* mainLayout = new QVBoxLayout(ui->groupBox_productType);
    ui->groupBox_productType->setLayout(mainLayout);

A layoutot a Designer adja, nem a kód.

3. LÉPÉS — A typeContainerWidget‑be tedd a QFlowLayout‑ot

A konstruktorban:

    ui->typeContainerWidget->setLayout(typeLayout);

Majd a típus rádiókat így generáld:

    typeLayout->addWidget(rb);

✔ Nem keverednek a calcMode rádiókkal

6. LÉPÉS — A populateCalcModePanel() függvényt javítsd ki

Törlendő minden, ami ui->groupBox_calcMode‑ra hivatkozik.

A végleges verzió:

    void AddInputDialog::populateCalcModePanel()
    {
        QString typeCode = currentProductTypeCode();
        QString subtypeCode = currentProductSubtypeCode();

        QVector<SizeCalcMode> modes =
            ProductCalcModeRegistry::instance().getModes(typeCode, subtypeCode);

        // ha nincs mód → ürítjük a layoutot
        if (modes.isEmpty()) {
            if (_calcModeLayout) {
                QLayoutItem* item;
                while ((item = _calcModeLayout->takeAt(0)) != nullptr) {
                    if (item->widget())
                        item->widget()->deleteLater();
                    delete item;
                }
            }
            return;
        }

        SizeCalcMode defaultMode =
            ProductCalcModeRegistry::instance().getDefault(typeCode, subtypeCode);

        // régi tartalom törlése
        if (_calcModeLayout) {
            QLayoutItem* item;
            while ((item = _calcModeLayout->takeAt(0)) != nullptr) {
                if (item->widget())
                    item->widget()->deleteLater();
                delete item;
            }
        }

        // új rádiók
        for (SizeCalcMode m : modes) {
            auto* rb = new QRadioButton(SizeCalcModeUtils::toString(m),
                                        ui->calcModeContainerWidget);
            rb->setProperty("calcMode", static_cast<int>(m));
            _calcModeLayout->addWidget(rb);

            if (m == defaultMode)
                rb->setChecked(true);
        }
    }

✔ A calcMode rádiók parentje calcModeContainerWidget

✔ Nem keverednek a type rádiókkal

✔ Nem keverednek az altípus rádiókkal

✔ A layout nem parent

✔ A UI stabil

7. LÉPÉS — A applyCalcModeFromRequest() függvényt javítsd ki

    void AddInputDialog::applyCalcModeFromRequest(const Cutting::Plan::Request& req)
    {
        SizeCalcMode mode = req.calcMode;

        if (!_calcModeLayout)
            return;

        for (int i = 0; i < _calcModeLayout->count(); ++i) {
            QWidget* w = _calcModeLayout->itemAt(i)->widget();
            if (!w) continue;

            auto* rb = qobject_cast<QRadioButton*>(w);
            if (!rb) continue;

            SizeCalcMode m =
                static_cast<SizeCalcMode>(rb->property("calcMode").toInt());

            rb->setChecked(m == mode);
        }
    }

8. LÉPÉS — A selectedCalcMode() függvényt javítsd ki

    SizeCalcMode AddInputDialog::selectedCalcMode() const
    {
        if (!_calcModeLayout)
            return SizeCalcMode::Unknown;

        for (int i = 0; i < _calcModeLayout->count(); ++i) {
            QWidget* w = _calcModeLayout->itemAt(i)->widget();
            if (!w) continue;

            auto* rb = qobject_cast<QRadioButton*>(w);
            if (!rb) continue;

            if (rb->isChecked()) {
                return static_cast<SizeCalcMode>(rb->property("calcMode").toInt());
            }
        }

        return SizeCalcMode::Unknown;
    }
